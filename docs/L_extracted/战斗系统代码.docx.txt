// ==================== 工具函数 ====================
function getLivePlayers() {
    return State.variables.party.filter(p => p.hp > 0);
}
function getLiveEnemies() {
    return State.variables.enemies.filter(e => e.hp > 0);
}
function getLowestHpTarget(players) {
    let lowest = null, minHp = Infinity;
    for (let p of players) {
        if (p.hp > 0 && p.hp < minHp) {
            minHp = p.hp;
            lowest = p;
        }
    }
    return lowest;
}
function deepCopy(obj) {
    return JSON.parse(JSON.stringify(obj));
}

// ==================== 战斗核心 ====================
window.generateTurnOrder = function() {
    let party = State.variables.party.filter(p => p.hp > 0);
    let enemies = State.variables.enemies.filter(e => e.hp > 0);
    let order = [...party, ...enemies];
    order.sort((a,b) => b.speed - a.speed);
    State.variables.turnOrder = order;
    State.variables.turnIndex = 0;
};

window.advanceTurn = function() {
    let turnOrder = State.variables.turnOrder;
    let idx = State.variables.turnIndex;
    do { idx++; } while (idx < turnOrder.length && turnOrder[idx].hp <= 0);
    if (idx >= turnOrder.length) {
        applyPassives();
        generateTurnOrder();
    } else {
        State.variables.turnIndex = idx;
    }
    checkBattleEnd();
    Engine.show("战斗");
};

function applyPassives() {
    let party = State.variables.party;
    for (let p of party) {
        if (p.hp <= 0) continue;
        if (p.name === "K" && p.form === "mage") {
            p.hp = Math.min(p.maxHp, p.hp + 2);
        }
        if (p.bless && p.bless.turns > 0) {
            p.hp = Math.min(p.maxHp, p.hp + p.maxHp * 0.005);
            p.mp = Math.min(p.maxMp, p.mp + p.maxMp * 0.005);
            p.bless.turns--;
            if (p.bless.turns <= 0) p.bless = null;
        }
        if (p.buff && p.buff.turns > 0) {
            p.buff.turns--;
            if (p.buff.turns <= 0) p.buff = { power:0, insight:0, dmgMult:0, turns:0 };
        }
        if (p.skillCooldown && p.skillCooldown.lightShield > 0) {
            p.skillCooldown.lightShield--;
        }
        if (p.rotTurns > 0) {
            p.rotTurns--;
            if (p.rotTurns <= 0) p.rotTurns = 0;
        }
    }
    let enemies = State.variables.enemies;
    for (let e of enemies) {
        if (e.buffTurns > 0) {
            e.buffTurns--;
            if (e.buffTurns <= 0) { e.buffDef = 0; e.buffStr = 0; }
        }
        if (e.burn && e.burn.turns > 0) {
            e.burn.turns--;
            if (e.burn.turns <= 0) e.burn.mult = 0;
        }
        if (e.frost && e.frost.turns > 0) {
            e.frost.turns--;
            if (e.frost.turns <= 0) e.frost.reduction = 0;
        }
    }
}

function checkBattleEnd() {
    let party = State.variables.party;
    let enemies = State.variables.enemies;
    let allEnemiesDead = enemies.every(e => e.hp <= 0);
    let partyAllDead = party.every(p => p.hp <= 0);
    if (allEnemiesDead) {
        State.variables.inCombat = false;
        let totalExp = 0, totalGold = 0;
        for (let e of enemies) {
            if (e.hp <= 0) {
                totalExp += e.expValue || 0;
                totalGold += e.gold || 0;
            }
        }
        for (let p of party) {
            if (p.hp > 0) {
                p.exp += totalExp;
                State.variables.gold += totalGold;
                while (p.exp >= p.expNext) {
                    p.exp -= p.expNext;
                    p.level++;
                    p.attrPoints++;
                    p.hp = Math.min(p.maxHp, p.hp + Math.floor(p.maxHp / 2));
                    p.mp = Math.min(p.maxMp, p.mp + Math.floor(p.maxMp / 2));
                    if (p.level === 5 || p.level === 10) {
                        p.atkBase += 3;
                        p.matkBase += 2;
                    }
                    p.expNext = Math.floor(p.expNext * 1.2);
                }
            }
        }
        if (enemies.some(e => e.isBoss)) {
            State.variables.forestLayer = 6;
        } else {
            State.variables.forestLayer += 1;
        }
        Engine.show("森林入口");
    } else if (partyAllDead) {
        State.variables.inCombat = false;
        Engine.show("队伍全灭");
    }
}

// ==================== 敌人技能库 ====================
window.enemySkills = {
    attack: function(caster, target, party) {
        let physRaw = caster.atkBase + 2 * (caster.power + (caster.buffStr || 0));
        let physDmg = Math.max(1, physRaw - target.def);
        let magRaw = caster.matkBase + 2 * caster.insight;
        let magDmg = Math.max(1, magRaw * (1 - target.mdef));
        if (physDmg >= magDmg) {
            target.hp -= physDmg;
            return `敌人使用冲撞，造成 ${physDmg.toFixed(1)} 点伤害。`;
        } else {
            target.hp -= magDmg;
            return `敌人使用暗影球，造成 ${magDmg.toFixed(1)} 点伤害。`;
        }
    },
    harden: function(caster, target, party) {
        caster.buffDef = (caster.buffDef || 0) + 2;
        caster.buffStr = (caster.buffStr || 0) + 1;
        caster.buffTurns = 2;
        return "敌人使用硬化，防御+2，力量+1，持续2回合。";
    },
    smash: function(caster, target, party) {
        caster.def = Math.max(0, caster.def - 1);
        let raw = caster.atkBase + 4 * (caster.power + (caster.buffStr || 0));
        let damage = Math.max(1, raw - target.def);
        target.hp -= damage;
        return `敌人使用粉身碎骨，造成 ${damage.toFixed(1)} 点伤害，自身防御降低1。`;
    },
    corrosion: function(caster, target, party) {
        let totalMsg = "";
        for (let p of party) {
            if (p.hp > 0) {
                let dmg = 10 * (1 - p.mdef);
                dmg = Math.max(1, Math.floor(dmg));
                p.hp -= dmg;
                p.rotTurns = 2;
                totalMsg += `${p.name} 受到 ${dmg} 点魔法伤害，施加腐烂。`;
            }
        }
        return "敌人使用腐蚀气息，" + totalMsg;
    }
};

// ==================== 敌人AI行动 ====================
window.enemyAITurn = function(enemy) {
    let party = State.variables.party;
    let msg = "";
    if (enemy.burn && enemy.burn.turns > 0) {
        let burnDmg = enemy.burn.mult * party[0].insight;
        enemy.hp -= burnDmg;
        msg += `灼伤造成 ${burnDmg.toFixed(1)} 伤害。`;
        enemy.burn.turns--;
        if (enemy.burn.turns <= 0) enemy.burn.mult = 0;
    }
    let target = getLowestHpTarget(party);
    if (!target) { window.advanceTurn(); return; }
    if (enemy.hp < enemy.maxHp / 2 && !enemy.desperateUsed) {
        enemy.desperateUsed = true;
        let skillName = enemy.emergencySkill || "smash";
        if (window.enemySkills[skillName]) {
            msg += window.enemySkills[skillName](enemy, target, party);
        }
        State.variables.lastMessage = msg;
        window.advanceTurn();
        return;
    }
    let cycle = enemy.actionCycle || 0;
    let skillName = enemy.skillCycle[cycle % enemy.skillCycle.length];
    if (window.enemySkills[skillName]) {
        msg += window.enemySkills[skillName](enemy, target, party);
    }
    enemy.actionCycle = (cycle + 1) % enemy.skillCycle.length;
    State.variables.lastMessage = msg;
    window.advanceTurn();
};

// ==================== 玩家角色行动 ====================
window.actorAct = function(actor, skill) {
    let party = State.variables.party;
    let enemies = State.variables.enemies;
    let msg = "";
    function getSingleEnemy() { return enemies.find(e => e.hp > 0); }
    function getAllEnemies() { return enemies.filter(e => e.hp > 0); }
    let power = actor.power + (actor.buff?.power || 0);
    let insight = actor.insight + (actor.buff?.insight || 0);
    let dmgMult = 1 + (actor.buff?.dmgMult || 0);
    // 勇者技能
    if (actor.name === "勇者") {
        if (skill === "light") {
            let target = getSingleEnemy();
            if (!target) return;
            let enemyDef = target.def + (target.buffDef || 0);
            let damage = Math.max(1, (actor.atkBase + power) - enemyDef);
            damage = Math.floor(damage * dmgMult);
            target.hp -= damage;
            msg = `勇者轻攻击造成 ${damage.toFixed(1)} 伤害。`;
        }
        else if (skill === "heavy") {
            let target = getSingleEnemy();
            if (!target) return;
            let cost = Math.max(1, Math.floor(actor.hp / 10));
            actor.hp -= cost;
            let enemyDef = target.def + (target.buffDef || 0);
            let effectiveDef = enemyDef * 0.8;
            let damage = Math.max(1, (2 * actor.atkBase + 2 * power) - effectiveDef);
            damage = Math.floor(damage * dmgMult);
            target.hp -= damage;
            msg = `重攻击消耗 ${cost} HP，造成 ${damage.toFixed(1)} 伤害。`;
        }
        else if (skill === "fire") {
            if (actor.mp < 5) { msg = "MP不足！"; } else {
                actor.mp -= 5;
                let target = getSingleEnemy();
                if (!target) return;
                let baseDmg = actor.matkBase + insight;
                let totalDmg = baseDmg + 2 * insight;
                totalDmg = totalDmg * (1 - target.mdef);
                totalDmg = Math.max(1, Math.floor(totalDmg * dmgMult));
                target.hp -= totalDmg;
                if (target.burn.turns > 0) target.burn.mult += 1;
                else target.burn.mult = 1;
                target.burn.turns = 3;
                msg = `小火球造成 ${totalDmg.toFixed(1)} 伤害，施加灼伤。`;
            }
        }
        else if (skill === "ice") {
            if (actor.mp < 5) { msg = "MP不足！"; } else {
                actor.mp -= 5;
                let target = getSingleEnemy();
                if (!target) return;
                let baseDmg = actor.matkBase + insight;
                let totalDmg = baseDmg + 1 * insight;
                totalDmg = totalDmg * (1 - target.mdef);
                totalDmg = Math.max(1, Math.floor(totalDmg * dmgMult));
                target.hp -= totalDmg;
                if (target.frost.turns > 0) target.frost.reduction += 0.1;
                else target.frost.reduction = 0.1;
                target.frost.turns = 2;
                msg = `冰刃造成 ${totalDmg.toFixed(1)} 伤害，施加冻伤。`;
            }
        }
        else if (skill === "powerstance") {
            if (actor.hp <= 5 || actor.mp < 5) { msg = "HP或MP不足！"; } else {
                actor.hp -= 5;
                actor.mp -= 5;
                actor.buff.power += 1;
                actor.buff.insight += 1;
                actor.buff.dmgMult += 0.2;
                actor.buff.turns = 3;
                msg = "全力架势：力量+1，灵感+1，伤害+20%，持续3回合。";
            }
        }
        else if (skill === "heal") {
            if (State.variables.healUsedToday >= State.variables.maxHealPerDay) {
                msg = "今日恢复次数已用完！";
            } else {
                State.variables.healUsedToday++;
                let target = party.reduce((min, p) => p.hp > 0 && p.hp < min.hp ? p : min, party[0]);
                let heal = Math.ceil(target.maxHp * 0.3);
                target.hp = Math.min(target.maxHp, target.hp + heal);
                msg = `恢复魔法治疗 ${target.name}，恢复 ${heal} HP。剩余次数 ${State.variables.maxHealPerDay - State.variables.healUsedToday}`;
            }
        }
    }
    // K技能
    else if (actor.name === "K") {
        if (skill === "form_mage") {
            actor.form = "mage";
            msg = "K切换到魔法元祖形态。";
        }
        else if (skill === "form_sword") {
            actor.form = "sword";
            msg = "K切换到剑术大师形态。";
        }
        else if (skill === "enchant_light") {
            if (actor.mp < 10) { msg = "MP不足！"; } else {
                actor.mp -= 10;
                for (let p of party) {
                    if (p.hp > 0) {
                        p.hp = Math.min(p.maxHp, p.hp + Math.ceil(p.maxHp * 0.02));
                    }
                }
                actor.power += 1;
                actor.enchant = { type: "light", turns: 4, extra: 1 };
                msg = "光辉于我：全体恢复2% HP，力量+1，普攻附加1倍灵感魔法伤害，持续4回合。";
            }
        }
        else if (skill === "enchant_fire") {
            if (actor.mp < 10) { msg = "MP不足！"; } else {
                actor.mp -= 10;
                let enemies = getAllEnemies();
                for (let e of enemies) {
                    if (e.burn.turns > 0) e.burn.mult += 1;
                    else e.burn.mult = 1;
                    e.burn.turns = 2;
                }
                actor.enchant = { type: "fire", turns: 4, extra: 1.5 };
                msg = "炙热于心：全体敌人受到2回合灼烧，普攻附加1.5倍灵感魔法伤害。";
            }
        }
        else if (skill === "enchant_ice") {
            if (actor.mp < 10) { msg = "MP不足！"; } else {
                actor.mp -= 10;
                actor.insight += 3;
                actor.mdef += 0.4;
                actor.enchant = { type: "ice", turns: 4, extra: 1 };
                msg = "严寒外衣：灵感+3，魔抗+40%，普攻附加冻伤，持续4回合。";
            }
        }
        else if (skill === "light_shield") {
            if (actor.skillCooldown.lightShield > 0) {
                msg = `光盾冷却中，剩余 ${actor.skillCooldown.lightShield} 回合。`;
            } else {
                actor.mp = Math.min(actor.maxMp, actor.mp + 2);
                actor.buff.defBonus = 2;
                actor.buff.mdefBonus = 0.1;
                actor.buff.dmgReduction = 0.3;
                actor.buff.turns = 2;
                actor.shield = { active: true, type: "percent", value: 0.5 };
                actor.skillCooldown.lightShield = 3;
                msg = "光盾：回复2MP，获得防御+2、魔抗+10%、减伤30%持续2回合，并获得一次50%减伤护盾。";
            }
        }
        else if (skill === "sword_single") {
            let target = getSingleEnemy();
            if (!target) return;
            let physDmg = actor.atkBase + 2 * power;
            let finalDmg = Math.max(1, physDmg - (target.def + (target.buffDef || 0)));
            if (actor.enchant && actor.enchant.turns > 0) {
                let extra = actor.enchant.extra * insight;
                finalDmg += extra;
                if (actor.enchant.type === "ice") {
                    if (target.frost.turns > 0) target.frost.reduction += 0.1;
                    else target.frost.reduction = 0.1;
                    target.frost.turns = 2;
                }
            }
            if (actor.form === "sword") finalDmg *= 1.1;
            finalDmg = Math.floor(finalDmg);
            target.hp -= finalDmg;
            msg = `K挥剑造成 ${finalDmg.toFixed(1)} 伤害。`;
            if (actor.enchant && actor.enchant.turns > 0) {
                actor.enchant.turns--;
                if (actor.enchant.turns <= 0) {
                    if (actor.enchant.type === "ice") {
                        actor.insight -= 3;
                        actor.mdef -= 0.4;
                    } else if (actor.enchant.type === "light") {
                        actor.power -= 1;
                    }
                    actor.enchant = { type: "", turns: 0, extra: 0 };
                }
            }
        }
        else if (skill === "sword_sweep") {
            let targets = getAllEnemies();
            if (targets.length === 0) return;
            let physBase = actor.atkBase + 2 * power;
            let physDmg = physBase * 0.6;
            for (let t of targets) {
                let dmg = Math.max(1, physDmg - (t.def + (t.buffDef || 0)));
                if (actor.enchant && actor.enchant.turns > 0) {
                    dmg += actor.enchant.extra * insight;
                    if (actor.enchant.type === "ice") {
                        if (t.frost.turns > 0) t.frost.reduction += 0.1;
                        else t.frost.reduction = 0.1;
                        t.frost.turns = 2;
                    }
                }
                if (actor.form === "sword") dmg *= 1.1;
                dmg = Math.floor(dmg);
                t.hp -= dmg;
            }
            msg = "K使用横扫，对所有敌人造成伤害。";
            if (actor.enchant && actor.enchant.turns > 0) {
                actor.enchant.turns--;
                if (actor.enchant.turns <= 0) {
                    if (actor.enchant.type === "ice") {
                        actor.insight -= 3;
                        actor.mdef -= 0.4;
                    } else if (actor.enchant.type === "light") {
                        actor.power -= 1;
                    }
                    actor.enchant = { type: "", turns: 0, extra: 0 };
                }
            }
        }
        else if (skill === "fireball") {
            if (actor.mp < 15) { msg = "MP不足！"; } else {
                actor.mp -= 15;
                let target = getSingleEnemy();
                if (!target) return;
                let base = actor.matkBase + insight;
                let dmg = base + 2 * insight;
                if (actor.form === "mage") dmg *= 1.2;
                dmg = dmg * (1 - target.mdef);
                dmg = Math.max(1, Math.floor(dmg));
                target.hp -= dmg;
                msg = `K小火球造成 ${dmg.toFixed(1)} 伤害。`;
            }
        }
        else if (skill === "holy_light") {
            if (actor.mp < 15) { msg = "MP不足！"; } else {
                actor.mp -= 15;
                for (let p of party) {
                    if (p.hp > 0) {
                        if (!p.bless) p.bless = { turns: 0 };
                        p.bless.turns = 3;
                    }
                }
                msg = "神圣光辉：全体队友获得祝福，每回合恢复0.5% HP/MP，持续3回合。";
            }
        }
    }
    State.variables.lastMessage = msg;
    window.advanceTurn();
};

// ==================== 初始化敌人 ====================
window.initEnemy = function(template) {
    let enemy = deepCopy(template);
    enemy.buffDef = 0;
    enemy.buffStr = 0;
    enemy.buffTurns = 0;
    enemy.burn = { turns:0, mult:0 };
    enemy.frost = { turns:0, reduction:0 };
    enemy.actionCycle = 0;
    enemy.desperateUsed = false;
    return enemy;
};
